// INTEGRAÇÃO API MERCADO LIVRE const APPS_SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbzyUBTwiKQptFE4BKvYs7RxuNXcZs3hM-ji4rLwsS-Av3nT812fXxQjok4Kc3cU35sVSw/exec'; const ML_AUTH_URL = 'https://auth.mercadolivre.com.br/authorization?response_type=code&client_id=4369092053425585&redirect_uri=https://betod50.github.io/Calculadora/'; let mlToken = null; async function initML() { const params = new URLSearchParams(window.location.search); const code = params.get('code'); if (code) { // Limpa o ?code= da URL sem recarregar a página window.history.replaceState({}, document.title, window.location.pathname); await trocarCodePorToken(code); } else { await obterTokenSalvo(); } } async function trocarCodePorToken(code) { try { mostrarStatusML('🔄 Conectando ao Mercado Livre...'); const res = await fetch(`${APPS_SCRIPT_URL}?action=exchange&code=${encodeURIComponent(code)}`); const data = await res.json(); if (data.access_token) { mlToken = data.access_token; mostrarStatusML('✅ Conectado ao ML'); await carregarDadosML(); } else { mostrarStatusML('❌ Erro ao conectar. Clique para autorizar novamente'); } } catch(e) { mostrarStatusML('❌ Falha na conexão com o servidor.'); } } async function obterTokenSalvo() { try { const res = await fetch(`${APPS_SCRIPT_URL}?action=token`); const data = await res.json(); if (data.access_token) { mlToken = data.access_token; mostrarStatusML('✅ ML conectado'); await carregarDadosML(); } else { mostrarStatusML('🔑 Clique aqui para conectar ao Mercado Livre'); } } catch(e) { mostrarStatusML('🔑 Conectar ao Mercado Livre'); } } async function carregarDadosML() { if (!mlToken) return; try { // Busca dados do usuário para confirmar conexão const res = await fetch('https://api.mercadolibre.com/users/me', { headers: { 'Authorization': 'Bearer ' + mlToken } }); const user = await res.json(); if (user.id) { mostrarStatusML(`✅ ML: ${user.nickname}`); } } catch(e) { console.log('Erro ao carregar dados ML:', e); } } function mostrarStatusML(msg) { let el = document.getElementById('ml-status'); if (!el) { el = document.createElement('div'); el.id = 'ml-status'; el.style.cssText = 'text-align:center;font-size:12px;padding:8px;background:#141824;border:1px solid #1a1f2e;border-radius:8px;margin-bottom:14px;color:#6b7280;'; document.querySelector('.container').insertBefore(el, document.querySelector('.plat-tabs')); } el.innerHTML = msg; } // Inicia integração ML ao carregar a página window.addEventListener('DOMContentLoaded', initML);
